# 导出模块实现

**版本**: 2.0  
**最后更新**: 2026-01-18

## 1. 概述

导出模块负责将浏览器端的一维卡片列表转换为 Obsidian Canvas 标准的二维网格布局。

### 1.1 核心流程

```
IndexedDB 读取
    ↓
CanvasNode[] (一维列表)
    ↓
List-to-Grid 转换算法
    ↓
ObsidianCanvasNode[] (二维网格)
    ↓
JSZip 打包
├── project.canvas (JSON 文件)
└── attachments/
    ├── image-1.png
    └── image-2.png
    ↓
浏览器下载 ZIP
```

## 2. JSON Canvas 规范

### 2.1 文件结构

```json
{
  "nodes": [
    {
      "id": "uuid-1",
      "type": "text",
      "x": 0,
      "y": 0,
      "width": 400,
      "height": 200,
      "text": "Hello World"
    },
    {
      "id": "uuid-2",
      "type": "file",
      "x": 450,
      "y": 0,
      "width": 400,
      "height": 300,
      "file": "attachments/image.png"
    },
    {
      "id": "uuid-3",
      "type": "link",
      "x": 0,
      "y": 250,
      "width": 400,
      "height": 150,
      "url": "https://example.com"
    }
  ],
  "edges": []
}
```

### 2.2 节点类型规范

| 类型 | 必需字段 | 可选字段 | 说明 |
|------|----------|----------|------|
| `text` | text | - | Markdown 文本 |
| `file` | file | - | 相对路径指向附件 |
| `link` | url | - | **只能有 url 字段，不能有 text** |
| `group` | - | label, color | 分组容器 |

### 2.3 坐标规范

- `x`, `y`, `width`, `height` 必须为**整数**
- 坐标单位：像素
- 原点 (0, 0) 在画布左上角

### 2.4 ID 规范

- 必须唯一
- 推荐使用 UUID 格式
- 示例：`a1b2c3d4-e5f6-7g8h-9i0j-k1l2m3n4o5p6`

## 3. List-to-Grid 算法

### 3.1 配置参数

```typescript
// src/sidepanel/services/exporter.ts

const CONFIG = {
  COLS: 4,           // 4 列网格布局
  CARD_WIDTH: 400,   // 卡片标准宽度 (像素)
  GAP: 50,           // 卡片间距 (像素)
  HEIGHT_TEXT: 200,  // 文本卡片高度
  HEIGHT_IMAGE: 300, // 图片卡片高度
  HEIGHT_LINK: 150   // 链接卡片高度
}
```

### 3.2 核心算法

```typescript
function convertToGrid(nodes: CanvasNode[]): ObsidianCanvasNode[] {
  // 1. 按 order 排序
  const sorted = nodes.sort((a, b) => a.order - b.order)

  // 2. 转换为 Obsidian Canvas 节点
  return sorted.map((node, index) => {
    // 一维索引转二维坐标
    const col = index % CONFIG.COLS
    const row = Math.floor(index / CONFIG.COLS)

    const x = Math.round(col * (CONFIG.CARD_WIDTH + CONFIG.GAP))
    const y = Math.round(row * (CONFIG.HEIGHT_TEXT + CONFIG.GAP))
    const height = estimateHeight(node.type)

    return {
      id: node.id,
      type: node.type,
      x,
      y,
      width: CONFIG.CARD_WIDTH,
      height
    }
  })
}

function estimateHeight(type: NodeType): number {
  switch (type) {
    case 'text': return CONFIG.HEIGHT_TEXT
    case 'file': return CONFIG.HEIGHT_IMAGE
    case 'link': return CONFIG.HEIGHT_LINK
  }
}
```

### 3.3 字段映射

```typescript
function buildObsidianNode(
  node: CanvasNode,
  x: number,
  y: number,
  height: number
): ObsidianCanvasNode {
  const base = {
    id: node.id,
    x, y,
    width: CONFIG.CARD_WIDTH,
    height
  }

  switch (node.type) {
    case 'text':
      return {
        ...base,
        type: 'text',
        text: node.text || ''
      }

    case 'file':
      return {
        ...base,
        type: 'file',
        file: `attachments/${node.fileName}`
      }

    case 'link':
      // ⚠️ 关键：Link 节点只能有 url 字段
      // 添加 text 字段会导致 Obsidian 解析错误
      return {
        ...base,
        type: 'link',
        url: node.url || ''
      }
  }
}
```

## 4. ZIP 打包实现

### 4.1 完整导出函数

```typescript
// src/sidepanel/services/exporter.ts

import JSZip from 'jszip'
import { saveAs } from 'file-saver'

export async function exportProject(projectId: string, projectName: string) {
  // 1. 从数据库读取所有节点
  const nodes = await db.nodes
    .where('projectId')
    .equals(projectId)
    .sortBy('order')

  if (nodes.length === 0) {
    throw new Error('项目为空，无法导出')
  }

  // 2. 创建 ZIP
  const zip = new JSZip()

  // 3. 创建 attachments 文件夹
  const attachmentsFolder = zip.folder('attachments')

  // 4. 转换节点
  const obsidianNodes: ObsidianCanvasNode[] = []

  for (const node of nodes) {
    const col = obsidianNodes.length % CONFIG.COLS
    const row = Math.floor(obsidianNodes.length / CONFIG.COLS)

    const x = Math.round(col * (CONFIG.CARD_WIDTH + CONFIG.GAP))
    const y = Math.round(row * (CONFIG.HEIGHT_TEXT + CONFIG.GAP))
    const height = estimateHeight(node.type)

    const obsidianNode = buildObsidianNode(node, x, y, height)
    obsidianNodes.push(obsidianNode)

    // 如果是图片节点，添加到 ZIP
    if (node.type === 'file' && node.fileData && node.fileName) {
      attachmentsFolder?.file(node.fileName, node.fileData)
    }
  }

  // 5. 生成 Canvas JSON
  const canvasData: ObsidianCanvasFile = {
    nodes: obsidianNodes,
    edges: []
  }

  // 6. 添加 .canvas 文件到 ZIP
  zip.file(`${projectName}.canvas`, JSON.stringify(canvasData, null, 2))

  // 7. 生成并下载
  const content = await zip.generateAsync({ type: 'blob' })
  saveAs(content, `${projectName}.zip`)

  console.log('[WebCanvas] Export completed:', projectName)
}
```

### 4.2 ZIP 文件结构

```
project-name.zip
├── project-name.canvas          # JSON Canvas 文件 (根目录)
└── attachments/                 # 附件目录
    ├── image-1737123456.png
    ├── image-1737123457.png
    └── image-1737123458.png
```

### 4.3 Obsidian 导入步骤

1. 下载 `.zip` 文件
2. 解压到 Obsidian Vault
3. 打开 `.canvas` 文件
4. 所有节点按 4 列网格自动排列
5. 图片自动加载（相对路径 `attachments/xxx.png`）

## 5. 已修复的问题

### 5.1 Link 节点 text 字段问题 ✅

**问题**：原代码在 Link 节点中添加了 `text` 字段，导致 Obsidian 解析错误。

**修复**：
```typescript
// ❌ 修复前
} else if (node.type === 'link') {
  obsidianNode.url = node.url || ''
  obsidianNode.text = node.text || node.url || ''  // 错误！
}

// ✅ 修复后
} else if (node.type === 'link') {
  // Link 节点只能有 url 字段
  obsidianNode.url = node.url || ''
}
```

### 5.2 节点高度过小 ✅

**问题**：原高度设置导致内容被截断。

**修复**：
```typescript
// 修复前
const CARD_HEIGHT_TEXT = 120
const CARD_HEIGHT_IMAGE = 200
const CARD_HEIGHT_LINK = 100

// 修复后
const CARD_HEIGHT_TEXT = 200    // +80px

```

### 5.3 坐标未取整 ✅

**问题**：JSON Canvas 规范要求坐标为整数。

**修复**：
```typescript
// 使用 Math.round() 确保整数
const x = Math.round(col * (CARD_WIDTH + GAP))
const y = Math.round(row * (height + GAP))
```

### 5.4 ID 格式优化 ✅

**改进**：使用标准 UUID 格式。

```typescript
function generateUUID(): string {
  return 'xxxxxxxx-xxxx-4xxx-yxxx-xxxxxxxxxxxx'.replace(/[xy]/g, (c) => {
    const r = Math.random() * 16 | 0
    const v = c === 'x' ? r : (r & 0x3 | 0x8)
    return v.toString(16)
  })
}
```
