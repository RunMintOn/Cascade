## 1. 智能拖拽系统

### 1.1 概述

系统支持两种拖拽场景：
1. **外部拖拽** - 从网页拖入侧边栏
2. **内部排序** - 侧边栏内卡片重新排序

### 1.2 外部拖拽实现

#### Content Script (注入网页)

```typescript
// src/content/drag-listener.ts

document.addEventListener('dragstart', (event) => {
  const target = event.target as HTMLElement
  const payload: DragPayload = {
    sourceUrl: window.location.href,
    sourceTitle: document.title,
    type: 'unknown',
    content: null
  }

  if (target.tagName === 'IMG') {
    payload.type = 'image'
    payload.src = target.src
  } else if (target.tagName === 'A') {
    payload.type = 'link'
    payload.href = target.href
    payload.text = target.innerText
  } else {
    const selection = window.getSelection()?.toString()
    if (selection) {
      payload.type = 'text'
      payload.content = selection
    }
  }

  // 使用自定义 MIME type 传递数据
  event.dataTransfer?.setData(
    'application/webcanvas-payload',
    JSON.stringify(payload)
  )
})
```

#### Side Panel (接收端)

```typescript
// src/sidepanel/components/common/DropZone.tsx

const handleDrop = async (e: React.DragEvent) => {
  e.preventDefault()
  
  const rawData = e.dataTransfer.getData('application/webcanvas-payload')
  
  if (rawData) {
    const payload: DragPayload = JSON.parse(rawData)
    await handlePayload(payload)
  } else {
    // Fallback: 处理系统级拖拽（如本地文件）
    const files = e.dataTransfer.files
    if (files.length > 0) {
      await handleFiles(files)
    }
  }
}

const handlePayload = async (payload: DragPayload) => {
  switch (payload.type) {
    case 'text':
      await addTextNode(payload.content!, payload.sourceUrl)
      break
    case 'image':
      await downloadAndSaveImage(payload.src!, payload.sourceUrl)
      break
    case 'link':
      await addLinkNode(payload.href!, payload.text, payload.sourceUrl)
      break
  }
}
```

### 1.3 内部排序实现 (@dnd-kit)

```typescript
// src/sidepanel/components/CardStream/index.tsx

import { DndContext, DragEndEvent } from '@dnd-kit/core'
import { SortableContext, verticalListSortingStrategy } from '@dnd-kit/sortable'

export function CardStream({ projectId }: { projectId: string }) {
  const nodes = useLiveQuery(() => 
    db.nodes.where('projectId').equals(projectId).sortBy('order')
  )

  const handleDragEnd = async (event: DragEndEvent) => {
    const { active, over } = event
    
    if (!over || active.id === over.id) return

    const oldIndex = nodes.findIndex(n => n.id === active.id)
    const newIndex = nodes.findIndex(n => n.id === over.id)
    
    // 重新排序
    const newNodes = [...nodes]
    const [removed] = newNodes.splice(oldIndex, 1)
    newNodes.splice(newIndex, 0, removed)
    
    // 更新 order 值
    await db.transaction('rw', db.nodes, async () => {
      for (let i = 0; i < newNodes.length; i++) {
        await db.nodes.update(newNodes[i].id, { order: i })
      }
    })
  }

  return (
    <DndContext onDragEnd={handleDragEnd}>
      <SortableContext items={nodes} strategy={verticalListSortingStrategy}>
        {nodes.map(node => (
          <SortableCard key={node.id} node={node} />
        ))}
      </SortableContext>
    </DndContext>
  )
}
```

## 2. 粘贴功能 (Ctrl+V)

### 2.1 事件监听

```typescript
// src/sidepanel/components/CardStream/index.tsx

const handlePaste = useCallback(async (e: React.ClipboardEvent) => {
  e.preventDefault()
  
  console.log('[WebCanvas] Paste event triggered')
  
  // 1. 处理纯文本
  const text = e.clipboardData.getData('text/plain')
  if (text && text.trim()) {
    console.log('[WebCanvas] Pasted text:', text.substring(0, 50))
    await addTextNode(text.trim(), window.location.href)
    return
  }

  // 2. 处理图片 (剪贴板文件)
  const items = e.clipboardData.items
  for (const item of items) {
    if (item.kind === 'file' && item.type.startsWith('image/')) {
      const file = item.getAsFile()
      if (file) {
        const fileName = `image-${Date.now()}.${file.type.split('/')[1]}`
        console.log('[WebCanvas] Pasted image:', fileName)
        await addImageNode(projectId, file, fileName, window.location.href)
        return
      }
    }
  }

  // 3. 处理 HTML (提取图片/链接)
  const html = e.clipboardData.getData('text/html')
  if (html) {
    const tempDiv = document.createElement('div')
    tempDiv.innerHTML = html
    
    const img = tempDiv.querySelector('img')
    if (img) {
      await downloadAndSaveImage(img.src, window.location.href)
      return
    }
    
    const link = tempDiv.querySelector('a')
    if (link) {
      await addLinkNode(link.href, link.textContent || '', window.location.href)
      return
    }
  }

  // 4. 处理 URL
  const url = e.clipboardData.getData('text/uri-list')
  if (url && url.startsWith('http')) {
    console.log('[WebCanvas] Pasted URL:', url)
    await addLinkNode(url, undefined, getFavicon(url))
  }
}, [projectId])
```

### 2.2 支持的粘贴类型

| 类型 | 检测方式 | 处理逻辑 |
|------|----------|----------|
| 纯文本 | `text/plain` | 创建 Text 节点 |
| 剪贴板图片 | `items[].kind === 'file'` | 直接保存文件 |
| HTML 图片 | `text/html` → `<img>` | 后台下载图片 |
| 链接 | `text/uri-list` | 创建 Link 节点 |
| HTML 链接 | `text/html` → `<a>` | 创建 Link 节点 |

## 3. 图片处理

### 3.1 Background Service Worker

```typescript
// src/background/service-worker.ts

chrome.runtime.onMessage.addListener((message, sender, sendResponse) => {
  if (message.action === 'downloadImage') {
    downloadImage(message.url, message.projectId, message.sourceUrl)
      .then(result => sendResponse(result))
      .catch(error => sendResponse({ error: error.message }))
    
    return true // 保持消息通道开放
  }
})

async function downloadImage(url: string, projectId: string, sourceUrl: string) {
  try {
    const response = await fetch(url, {
      method: 'GET',
      credentials: 'omit', // 避免 CORS 问题
      headers: {
        'User-Agent': 'Mozilla/5.0 (WebCanvas Extension)'
      }
    })
    
    if (!response.ok) throw new Error(`HTTP ${response.status}`)
    
    const blob = await response.blob()
    const fileName = `image-${Date.now()}.${getFileNameFromUrl(url)}`
    
    // 存储到 IndexedDB
    await db.nodes.add({
      id: generateUUID(),
      projectId,
      type: 'file',
      fileData: blob,
      fileName,
      sourceUrl,
      createdAt: Date.now(),
      order: await getMaxOrder(projectId) + 1
    })
    
    console.log('[WebCanvas] Image downloaded:', fileName)
    return { success: true, fileName }
  } catch (error) {
    console.error('[WebCanvas] Download failed:', error)
    throw error
  }
}
```

### 3.2 前端渲染

```typescript
// src/sidepanel/components/CardStream/ImageCard.tsx

export function ImageCard({ node }: { node: CanvasNode }) {
  const [imageUrl, setImageUrl] = useState<string>('')
  const [loading, setLoading] = useState(true)
  const [error, setError] = useState(false)

  useEffect(() => {
    if (!node.fileData) return

    const url = URL.createObjectURL(node.fileData)
    setImageUrl(url)
    setLoading(false)

    return () => {
      URL.revokeObjectURL(url) // 清理内存
    }
  }, [node.fileData])

  if (loading) {
    return <div className="skeleton animate-pulse h-48 bg-gray-200" />
  }

  if (error || !imageUrl) {
    return (
      <div className="error-state p-4 text-center text-gray-500">
        <Icon name="image-off" />
        <p>加载失败</p>
      </div>
    )
  }

  return (
    <img
      src={imageUrl}
      alt={node.fileName}
      className="w-full h-48 object-cover rounded-lg"
    />
  )
}
```

## 5. 删除与撤销

### 5.1 删除逻辑

```typescript
// src/sidepanel/hooks/useDeleteNode.ts

export function useDeleteNode(projectId: string) {
  const [deletedNode, setDeletedNode] = useState<CanvasNode | null>(null)
  const [undoTimer, setUndoTimer] = useState<NodeJS.Timeout | null>(null)

  const deleteNode = async (nodeId: string) => {
    const node = await db.nodes.get(nodeId)
    if (!node) return

    // 从数据库删除
    await db.nodes.delete(nodeId)
    setDeletedNode(node)

    // 3 秒后自动清除撤销状态
    const timer = setTimeout(async () => {
      setDeletedNode(null)
    }, 3000)

    setUndoTimer(timer)
  }

  const undoDelete = async () => {
    if (!deletedNode || undoTimer) return

    clearTimeout(undoTimer)
    await db.nodes.add(deletedNode)
    setDeletedNode(null)
  }

  return { deleteNode, undoDelete, deletedNode }
}
```

```typescript
// src/sidepanel/components/common/DropZone.tsx

export function DropZone({ mode = 'normal', onDrop }: DropZoneProps) {
  const [isDragging, setIsDragging] = useState(false)

  const neonClass = mode === 'inbox' 
    ? 'animate-neon-breathe-green border-green-500' 
    : 'animate-neon-breathe border-blue-500'

  return (
    <>
      {isDragging && (
        <>
          {/* 光效层 - 顶层 */}
          <div className={`absolute inset-0 border-2 ${neonClass} z-100 pointer-events-none`} />
          
          {/* 半透明背景 */}
          <div className="absolute inset-0 bg-blue-500/10 z-100 pointer-events-none" />
          
          {/* 底部提示胶囊 */}
          <div className="fixed bottom-8 left-1/2 -translate-x-1/2 bg-gray-900/90 text-white px-6 py-3 rounded-full z-101">
            {mode === 'inbox' ? '释放保存到收集箱' : '释放以添加内容'}
          </div>
        </>
      )}
      
      <div
        onDragOver={(e) => { e.preventDefault(); setIsDragging(true) }}
        onDragLeave={() => setIsDragging(false)}
        onDrop={(e) => { setIsDragging(false); onDrop(e) }}
      />
    </>
  )
}
```
